cmake_minimum_required(VERSION 3.20)
project(StartApp VERSION 0.6.0 LANGUAGES C)

if(MSVC)
    message(FATAL_ERROR "Use the MSYS2 UCRT64 GCC toolchain on Windows; see docs/PLATFORMS.md")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(STARTAPP_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE BOOL "Export compiler arguments for project analysis")
# The integration test imports the actual build, not only hand-written JSON.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/.cmake/api/v1/query")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/.cmake/api/v1/query/codemodel-v2" "")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/.cmake/api/v1/query/toolchains-v1" "")
if(CMAKE_VERSION VERSION_GREATER_EQUAL 3.27)
    cmake_file_api(QUERY API_VERSION 1 CODEMODEL 2 TOOLCHAINS 1)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.12)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0>=2.68)
pkg_check_modules(JSON_GLIB REQUIRED IMPORTED_TARGET json-glib-1.0>=1.6)
pkg_check_modules(SOURCEVIEW REQUIRED IMPORTED_TARGET gtksourceview-5>=5.4)
find_program(GLIB_COMPILE_RESOURCES glib-compile-resources REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/startapp-resources.c"
    COMMAND "${GLIB_COMPILE_RESOURCES}" --generate-source --c-name startapp_resources
        --target "${CMAKE_CURRENT_BINARY_DIR}/startapp-resources.c"
        --sourcedir "${CMAKE_CURRENT_SOURCE_DIR}/resources" "${CMAKE_CURRENT_SOURCE_DIR}/resources/startapp.gresource.xml"
    DEPENDS resources/startapp.gresource.xml resources/styles/startapp-dark.xml)

add_library(startapp_options INTERFACE)
target_compile_options(startapp_options INTERFACE -Wall -Wextra -Wpedantic)
if(STARTAPP_WARNINGS_AS_ERRORS)
    target_compile_options(startapp_options INTERFACE -Werror)
endif()
add_library(startapp_core STATIC src/project.c src/document.c src/build_metadata.c src/graph_layout.c src/build.c src/session.c src/build_fixture.c src/workspace_tools.c src/extra_metadata.c src/language_client.c src/recovery.c src/snapshot.c)
target_include_directories(startapp_core PUBLIC src)
target_link_libraries(startapp_core PUBLIC PkgConfig::GIO PkgConfig::JSON_GLIB startapp_options)
add_library(startapp_ui STATIC src/app.c src/editor.c src/graph.c src/theme.c src/build_panel.c src/tools_ui.c src/language_ui.c "${CMAKE_CURRENT_BINARY_DIR}/startapp-resources.c")
target_link_libraries(startapp_ui PUBLIC startapp_core PkgConfig::GTK4 PkgConfig::SOURCEVIEW)
if(NOT WIN32)
    target_link_libraries(startapp_ui PUBLIC m)
endif()

add_executable(startapp src/main.c)
target_link_libraries(startapp PRIVATE startapp_ui)
if(WIN32)
    # Double-clicking the desktop executable must not open a console window.
    set_target_properties(startapp PROPERTIES WIN32_EXECUTABLE TRUE)
    add_executable(startapp-cli src/main.c)
    target_link_libraries(startapp-cli PRIVATE startapp_ui)
    set(STARTAPP_TEST_EXECUTABLE startapp-cli)
else()
    set(STARTAPP_TEST_EXECUTABLE startapp)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(startapp-core-tests tests/test_core.c)
    target_link_libraries(startapp-core-tests PRIVATE startapp_core)
    add_test(NAME core COMMAND startapp-core-tests)
    add_executable(startapp-tools-tests tests/test_tools.c)
    target_link_libraries(startapp-tools-tests PRIVATE startapp_core)
    add_test(NAME tools COMMAND startapp-tools-tests)
    set_tests_properties(tools PROPERTIES TIMEOUT 60)
    add_executable(startapp-workflow-tests tests/test_workflows.c)
    target_link_libraries(startapp-workflow-tests PRIVATE startapp_core)
    add_test(NAME workflows COMMAND startapp-workflow-tests)
    set_tests_properties(workflows PROPERTIES TIMEOUT 150)
    add_executable(startapp-build-tests tests/test_build_run.c)
    target_link_libraries(startapp-build-tests PRIVATE startapp_core)
    add_test(NAME build-run COMMAND startapp-build-tests)
    set_tests_properties(build-run PROPERTIES TIMEOUT 45)
    add_executable(startapp-cmake-build-tests tests/test_cmake_build.c)
    target_link_libraries(startapp-cmake-build-tests PRIVATE startapp_core)
    add_test(NAME cmake-build COMMAND startapp-cmake-build-tests "${CMAKE_COMMAND}")
    set_tests_properties(cmake-build PROPERTIES TIMEOUT 90)
    add_executable(startapp-import-tests tests/test_import.c)
    target_link_libraries(startapp-import-tests PRIVATE startapp_core)
    add_test(NAME metadata-self COMMAND ${CMAKE_COMMAND}
        "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}" "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}"
        "-DTEST_EXE=$<TARGET_FILE:startapp-import-tests>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_import.cmake")
    set_tests_properties(metadata-self PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
    add_test(NAME gui COMMAND ${STARTAPP_TEST_EXECUTABLE} --smoke-test)
    set_tests_properties(gui PROPERTIES TIMEOUT 45 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        ENVIRONMENT "GTK_A11Y=none;GSETTINGS_BACKEND=memory;GSK_RENDERER=cairo;STARTAPP_SMOKE_SCREENSHOT=${CMAKE_CURRENT_BINARY_DIR}/smoke-ui.png")
endif()

include(GNUInstallDirs)
install(TARGETS startapp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(WIN32)
    install(TARGETS startapp-cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
else()
    install(FILES packaging/io.github.PaperGooseRus.StartApp.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES packaging/io.github.PaperGooseRus.StartApp.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    set(CPACK_GENERATOR TGZ)
    set(CPACK_PACKAGE_NAME StartApp)
    set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
    set(CPACK_PACKAGE_FILE_NAME "StartApp-${PROJECT_VERSION}-Linux-${CMAKE_SYSTEM_PROCESSOR}")
    include(CPack)
endif()
